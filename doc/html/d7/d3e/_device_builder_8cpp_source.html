153void DeviceBuilder::buildDKDevice() {
154
155 // ----- Skills to Dynamic Vector (so it doesn't get deleted when builder dies) ------
156 std::vector<void(*)()>* dynamicSkillVector = new std::vector<void(*)()>;
157 dynamicSkillVector->assign(skillVector.begin(), skillVector.end());
158
159 // ----- SkillNames to Array (Optimizing data) ------
160 std::string *skillNamesArray = new std::string[skillNamesVector.size()];
161 for (int i = 0; i < this->skillNamesVector.size(); i++){
162 skillNamesArray[i] = skillNamesVector[i];
163 }
164
165 // ----- Building DataDevice ------
166 if (this->isDataDevice) {
167 DataDevice *dataDev = new DataDevice(this->name, this->id, dynamicSkillVector, skillNamesArray);
168
169 // Setting DataDevice limit (optional)
170 if (this->limit != nullptr)
171 (*dataDev).setLimit(this->limit);
172
173 (*dataDev).dataGenerator = this->dataGenerator;
174
175 // Saving Device Pointers
176 allDevices.push_back(*dataDev);
177 allDataDevices.push_back(*dataDev);
178
179 // ----- Building KernelDevice ------
180 }else {
181
182 KernelDevice* kernDev = new KernelDevice(this->name, this->id);
183
184 (*kernDev).securityGenerator = this->securityGenerator;
185
186 // Saving Device Pointers
187 allDevices.push_back(*kernDev);
188 allKernelDevices.push_back(*kernDev);
189 }
